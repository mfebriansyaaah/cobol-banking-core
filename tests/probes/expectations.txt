# Recorded detection capability of the static gate layer.
#
# Format: <probe>=<detected|missed>
#
#   detected -> at least one gate in tests/probes/run_probes.sh exits non-zero
#   missed   -> every gate passes on deliberately broken code (recorded gap)
#
# These are measurements, not aspirations. When a gate is added that closes a
# gap, the probe flips to "detected" and run_probes.sh fails until this file is
# updated on purpose. Do not lower an expectation to make the run green.

# Syntax: a route file that cannot be imported must stop the pipeline.
p1_broken_route_syntax=detected

# Two handlers for the same method+path; the second is unreachable.
p2_duplicate_route=detected

# One endpoint in a multi-route file loses tokenIdentity. check_route_auth
# walks route objects, so a single unguarded mutating endpoint is enough.
p3a_one_route_without_identity=detected

# Every route in a file loses tokenIdentity. Both the file-level guard and the
# per-route guard fire.
p3b_whole_route_file_without_identity=detected

# A route action reuses an existing dispatch prefix but no COBOL module
# implements it. route_dispatch (startsWith) accepts it; action_coverage
# notices the module has no such action.
p4_action_without_cobol_handler=detected

# A value moved into a WS-BIND-n that the following SQL call does not pass.
# This is the shipped P0-2 defect: the query is prepared with an empty
# parameter and silently matches nothing.
p5_dead_bind_slot=detected

# A true swap: the value for slot 1 goes to slot 2 and vice versa. Both slots
# are passed, both are valid, and only the query semantics are wrong. No static
# check can read the SQL author's intent; this needs a runtime suite.
p5b_swapped_bind_slots=missed

# A DB suite deleted outright. check_suite_coverage compares the directory
# against tests/baseline_counts.txt, so the inventory cannot shrink silently.
p6_deleted_db_suite=detected

# Five assertions removed from a suite that still exits 0. check_test_counts
# compares static assertion counts against the baseline, so a shrinking
# denominator is detected even though the runner stays green.
p7_missing_assertion=detected

# Two field names swapped in one route. Registration, auth, validation, and
# the COBOL handler all still exist; only the positional gateway->COBOL
# mapping shifts. check_field_order notices.
p8_field_order_drift=detected

# One run_cobol invocation replaced by a canned SUCCESS string. The suite
# still exits 0 with all assertions green and the counts unchanged, so neither
# the runner nor the count gate notices — but the action is never executed,
# which only check_action_test_coverage sees.
p9_action_without_test=detected

# A service reaching back into transport (services -> src). Only the
# dependency graph (ADR-0006 one-way layering) sees the direction.
p10_reverse_import=detected

# One field keeps its name but changes pattern (LIMIT -> INT). Names,
# registration, auth, and dispatch are untouched, so only the contract —
# the sole gate recording patterns — notices.
p11_pattern_swap=detected

# An orphan COMMIT in a module that never opens a transaction. Only the
# transaction-ownership gate pairs every close with an open.
p12_orphan_commit=detected

# A new COBOL->COBOL edge the registry never reviewed. Only the call-graph
# ratchet lists every cross-module edge.
p13_new_call_edge=detected

# A query buffer declared wider than the C bridge copies. Only the
# field-size gate enforces the 1024-byte ceiling.
p14_oversize_buffer=detected

# Trailing garbage that breaks YAML parsing. Only the openapi gate parses
# the file instead of string-comparing it.
p15_openapi_invalid=detected
